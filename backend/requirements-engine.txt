# Deep Reasoning (Ouroboros) engine deps — heavy LangGraph/LangChain stack,
# kept separate from the lean core runtime (requirements.txt) since the everyday
# chat engine needs none of it. Pinned to the versions Ouroboros is proven on.
langgraph==1.2.6
langchain==1.3.11
langchain-core==1.5.3
langchain-groq==1.1.3
langchain-community==0.4.2
groq==0.37.1
# Deep and agent runs follow the workspace's own provider, so they need a
# generic OpenAI-compatible client — Ollama's /v1, vLLM, LM Studio, OpenRouter.
# ChatGroq cannot stand in for this: the Groq SDK appends its own
# /openai/v1/chat/completions to any base_url, which 404s everywhere else.
langchain-openai==1.4.1
# Real semantic embeddings for convergence + memory (MiniLM). Without this the
# engine silently degrades to a lexical bag-of-words fallback — functional, but
# "semantic convergence" stops being semantic.
sentence-transformers==5.6.0
# Durable LangGraph checkpoints. Without this, a paused deep run (waiting for
# human guidance) or a paused agent turn (waiting for tool approval) lives in a
# process-local dict and dies on restart — see api/checkpointing.py, which
# falls back to memory and says so rather than refusing to start.
langgraph-checkpoint-sqlite==3.1.1
